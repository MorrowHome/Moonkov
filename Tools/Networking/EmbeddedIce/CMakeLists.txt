cmake_minimum_required(VERSION 3.21)
project(MoonEmbeddedIce LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if((NOT WIN32 AND NOT APPLE) OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Embedded ICE supports Windows x64 and macOS 64-bit.")
endif()
if(NOT EXISTS "${LIBJUICE_SOURCE}/include/juice/juice.h")
  message(FATAL_ERROR "Run Build.ps1 or Build-macOS.sh to fetch the pinned libjuice source.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_SERVER ON CACHE BOOL "" FORCE)
set(USE_NETTLE OFF CACHE BOOL "" FORCE)
set(ENABLE_LOCALHOST_ADDRESS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBJUICE_SOURCE}" libjuice EXCLUDE_FROM_ALL)
add_library(moon_ice SHARED moon_ice.cpp)
target_compile_definitions(moon_ice PRIVATE MOON_ICE_EXPORTS)
target_link_libraries(moon_ice PRIVATE juice-static)
add_executable(moon_ice_checks checks.cpp)
target_link_libraries(moon_ice_checks PRIVATE moon_ice)
if(MSVC)
  target_compile_options(moon_ice PRIVATE /W4 /WX /utf-8)
  target_compile_options(moon_ice_checks PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(moon_ice PRIVATE -Wall -Wextra -Werror)
  target_compile_options(moon_ice_checks PRIVATE -Wall -Wextra -Werror)
endif()
if(APPLE)
  set_target_properties(moon_ice PROPERTIES
    MACOSX_RPATH ON
    INSTALL_NAME_DIR "@rpath"
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES)
endif()
enable_testing()
add_test(NAME embedded_ice_native COMMAND moon_ice_checks)
set_tests_properties(embedded_ice_native PROPERTIES TIMEOUT 25)
